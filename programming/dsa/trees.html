Trees & Heaps

Data arranged in a hierarchy. Each step down a balanced tree throws away half the remaining work.

Tree in one sentence

A tree is data arranged like a family tree or a folder structure: one item at the top, each item pointing down at zero or more children, and no loops. Drawn in computing with the root at the top and the leaves at the bottom.

Vocabulary

TermMeaning
NodeOne item in the tree - a value plus references to its children
RootThe single node at the top, with no parent
Parent / childA node directly above / below another
LeafA node with no children
SubtreeAny node together with everything beneath it
Depth of a nodeHow many steps down from the root it sits
Height of the treeThe depth of its deepest node
Binary treeA tree where every node has at most two children (Left and Right)
BalancedEvery leaf is at roughly the same depth - the property that keeps operations O(log n)

The binary search tree

A binary search tree (BST) adds one rule to a binary tree, and that rule is what makes it useful:

The BST rule

For every node: everything in its left subtree is smaller, and everything in its right subtree is larger.

Searching for 60: it is bigger than 50 so go right; smaller than 70 so go left; found. Three comparisons instead of seven. Every step down discards half the remaining tree - the same halving that makes binary search O(log n).

A binary search tree in C#

public class TreeNode<T>
{
    public T Value { get; set; }
    public TreeNode<T>? Left { get; set; }
    public TreeNode<T>? Right { get; set; }

    public TreeNode(T value) => Value = value;
}

public class BinarySearchTree<T> where T : IComparable<T>
{
    private TreeNode<T>? _root;
    public int Count { get; private set; }

    // ---- Insert: walk down, then hang the new node off a null branch -------
    public void Insert(T value)
    {
        _root = InsertInto(_root, value);
        Count++;
    }

    private TreeNode<T>? InsertInto(TreeNode<T>? node, T value)
    {
        if (node is null) return new TreeNode<T>(value);   // found the empty slot

        int comparison = value.CompareTo(node.Value);
        if (comparison < 0) node.Left = InsertInto(node.Left, value);
        else                node.Right = InsertInto(node.Right, value);
        // (equal values go right here; a real implementation might reject them)

        return node;
    }

    // ---- Search: the whole point of a BST ---------------------------------
    public bool Contains(T value)
    {
        TreeNode<T>? current = _root;

        while (current is not null)
        {
            int comparison = value.CompareTo(current.Value);
            if (comparison == 0) return true;
            current = comparison < 0 ? current.Left : current.Right;   // half discarded
        }
        return false;
    }

    // ---- Minimum and maximum: as far left / right as you can go -----------
    public T Min()
    {
        if (_root is null) throw new InvalidOperationException("Tree is empty.");
        TreeNode<T> current = _root;
        while (current.Left is not null) current = current.Left;
        return current.Value;
    }

    public T Max()
    {
        if (_root is null) throw new InvalidOperationException("Tree is empty.");
        TreeNode<T> current = _root;
        while (current.Right is not null) current = current.Right;
        return current.Value;
    }

    public int Height() => HeightOf(_root);

    private int HeightOf(TreeNode<T>? node) =>
        node is null ? -1 : 1 + Math.Max(HeightOf(node.Left), HeightOf(node.Right));

    // ---- Traversals -------------------------------------------------------
    public List<T> InOrder()
    {
        var result = new List<T>();
        InOrderInto(_root, result);
        return result;
    }

    private void InOrderInto(TreeNode<T>? node, List<T> into)
    {
        if (node is null) return;
        InOrderInto(node.Left, into);     // everything smaller, first
        into.Add(node.Value);             // then this node
        InOrderInto(node.Right, into);    // then everything larger
    }

    public List<T> PreOrder()
    {
        var result = new List<T>();
        PreOrderInto(_root, result);
        return result;
    }

    private void PreOrderInto(TreeNode<T>? node, List<T> into)
    {
        if (node is null) return;
        into.Add(node.Value);             // this node first
        PreOrderInto(node.Left, into);
        PreOrderInto(node.Right, into);
    }

    public List<T> PostOrder()
    {
        var result = new List<T>();
        PostOrderInto(_root, result);
        return result;
    }

    private void PostOrderInto(TreeNode<T>? node, List<T> into)
    {
        if (node is null) return;
        PostOrderInto(node.Left, into);
        PostOrderInto(node.Right, into);
        into.Add(node.Value);             // this node last
    }

    // Level order needs a queue, not recursion
    public List<T> LevelOrder()
    {
        var result = new List<T>();
        if (_root is null) return result;

        var queue = new Queue<TreeNode<T>>();
        queue.Enqueue(_root);

        while (queue.Count > 0)
        {
            TreeNode<T> node = queue.Dequeue();
            result.Add(node.Value);
            if (node.Left is not null)  queue.Enqueue(node.Left);
            if (node.Right is not null) queue.Enqueue(node.Right);
        }
        return result;
    }
}
var tree = new BinarySearchTree<int>();
foreach (int n in new[] { 50, 30, 70, 20, 40, 60, 80 })
    tree.Insert(n);

Console.WriteLine(tree.Contains(60));                     // True
Console.WriteLine(tree.Contains(65));                     // False
Console.WriteLine($"{tree.Min()} .. {tree.Max()}");       // 20 .. 80
Console.WriteLine(tree.Height());                         // 2

Console.WriteLine(string.Join(" ", tree.InOrder()));      // 20 30 40 50 60 70 80
Console.WriteLine(string.Join(" ", tree.PreOrder()));     // 50 30 20 40 70 60 80
Console.WriteLine(string.Join(" ", tree.PostOrder()));    // 20 40 30 60 80 70 50
Console.WriteLine(string.Join(" ", tree.LevelOrder()));   // 50 30 70 20 40 60 80
In-order traversal gives you sorted output for free

"Left, me, right" on a BST always produces the values in ascending order. That is why SortedDictionary and SortedSet - both trees - can be iterated in sorted order with no sorting step.

Which traversal when

TraversalOrderUse it for
In-orderLeft, node, rightReading a BST in sorted order
Pre-orderNode, left, rightCopying a tree, serialising its structure
Post-orderLeft, right, nodeDeleting a tree, evaluating an expression tree, computing folder sizes
Level-order (BFS)Row by row, top to bottomPrinting by depth, finding the shallowest match

Balance is everything

A BST is only O(log n) while it stays bushy. Insert values that are already sorted and every node goes to the right - the tree becomes a linked list, and search degrades to O(n).

// Inserting 10, 20, 30, 40, 50 in order produces:
//   10
//     \
//      20
//        \
//         30
//           \
//            40
//              \
//               50
// Height 4 for 5 items. Contains(50) now costs 5 comparisons, not 3.

var degenerate = new BinarySearchTree<int>();
foreach (int n in new[] { 10, 20, 30, 40, 50 }) degenerate.Insert(n);
Console.WriteLine(degenerate.Height());   // 4 - a list wearing a tree costume

Self-balancing trees solve this by rotating nodes after each insert to keep the height near log n:

TreeIdeaWhere you meet it
AVL treeStrictly balanced; the two subtrees of any node differ in height by at most 1Read-heavy in-memory indexes
Red-black treeLooser balance, cheaper insertsSortedDictionary, SortedSet, java.util.TreeMap
B-tree / B+ treeMany keys per node, so each disk read fetches a lotSQL Server, PostgreSQL, and most database indexes
Trie (prefix tree)One node per characterAutocomplete, spell check, IP routing tables
 
// .NET's balanced trees - no need to write your own
var sorted = new SortedDictionary<string, int>();      // red-black tree
sorted["banana"] = 2;
sorted["apple"] = 5;
sorted["cherry"] = 1;

foreach (var pair in sorted)
    Console.WriteLine(pair.Key);        // apple, banana, cherry - always sorted

var set = new SortedSet<int> { 50, 10, 30 };
Console.WriteLine(string.Join(" ", set));                 // 10 30 50
Console.WriteLine(string.Join(" ", set.GetViewBetween(20, 60)));   // 30 50
Dictionary or SortedDictionary?

Dictionary is O(1) but unordered. SortedDictionary is O(log n) but iterates in key order and supports range queries. If you never need order, the hash table wins.

Heaps and priority queues

A heap is a different kind of binary tree with a weaker rule: every parent is smaller than its children (a min-heap). It does not sort the data - it only guarantees that the smallest item is at the root, which is exactly what a priority queue needs.

A heap is stored in a plain array, not with node objects: for the item at index i, its children live at 2i + 1 and 2i + 2, and its parent at (i - 1) / 2. No references, no allocations, excellent cache behaviour.

// .NET 6+ ships a heap-backed priority queue
var pq = new PriorityQueue<string, int>();
pq.Enqueue("low priority task", 5);
pq.Enqueue("urgent task", 1);
pq.Enqueue("normal task", 3);

Console.WriteLine(pq.Peek());        // urgent task        O(1)
Console.WriteLine(pq.Dequeue());     // urgent task        O(log n)
Console.WriteLine(pq.Dequeue());     // normal task

// Top-k without sorting everything: keep a heap of size k
IEnumerable<int> TopThree(IEnumerable<int> numbers)
{
    var heap = new PriorityQueue<int, int>();
    foreach (int n in numbers)
    {
        heap.Enqueue(n, n);
        if (heap.Count > 3) heap.Dequeue();   // drop the smallest
    }
    return heap.UnorderedItems.Select(x => x.Element).OrderByDescending(x => x);
}

Console.WriteLine(string.Join(" ", TopThree(new[] { 5, 1, 9, 3, 7, 2 })));  // 9 7 5

Trees that are not binary

Plenty of real trees have any number of children - file systems, org charts, category hierarchies, the DOM:

public class Category
{
    public string Name { get; }
    public List<Category> Children { get; } = new();

    public Category(string name) => Name = name;

    public Category Add(Category child)
    {
        Children.Add(child);
        return this;
    }

    // Recursive traversal with indentation for depth
    public void Print(int depth = 0)
    {
        Console.WriteLine(new string(' ', depth * 2) + Name);
        foreach (Category child in Children)
            child.Print(depth + 1);
    }

    // Post-order: children counted before the node itself
    public int CountAll() => 1 + Children.Sum(c => c.CountAll());
}

var root = new Category("Products");
var electronics = new Category("Electronics");
electronics.Add(new Category("Laptops")).Add(new Category("Phones"));
root.Add(electronics).Add(new Category("Books"));

root.Print();
// Products
//   Electronics
//     Laptops
//     Phones
//   Books

Console.WriteLine(root.CountAll());   // 5
